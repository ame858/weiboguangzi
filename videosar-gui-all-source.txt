"""Streamlit GUI for the Ka-band video SAR shadow project.

Start from the repository root:

    streamlit run app.py

The GUI is presentation-only: it shells out to ``python -m videosar`` and reads
existing result folders. It never modifies videosar/ or deliverables/.
"""

# Newer Streamlit releases renamed use_container_width to width='stretch'.
# Probe the signature once so the GUI works on both 1.37+ and 1.6x+.
_WIDTH_KWARG = ({"width": "stretch"}
                if "width" in inspect.signature(st.image).parameters
                else {"use_container_width": True})


def display_settings():
    """Shared display controls. Explicitly labelled as visual-only."""
    with st.sidebar.expander("显示增强（仅用于视觉增强，不改变底层检测输入）", expanded=True):
        mode = st.selectbox("灰度映射模式", viz.MODES,
                            format_func=lambda key: viz.MODE_LABELS[key], key="disp_mode")
        cmap = st.selectbox("伪彩色", viz.CMAPS, key="disp_cmap")
        denoise = st.checkbox("轻度显示去斑平滑（仅预览）", value=False, key="disp_denoise")
        diff_ref = st.selectbox("时序差分参考", ("median", "previous"),
                                format_func=lambda key: "短期历史中值" if key == "median" else "前一帧",
                                key="disp_diff")
        show_truth = st.checkbox("叠加绿色真值轮廓", value=True, key="disp_truth")
        show_pred = st.checkbox("叠加红色预测掩膜", value=True, key="disp_pred")
    return {"mode": mode, "cmap": cmap, "denoise": denoise, "diff_ref": diff_ref,
            "show_truth": show_truth, "show_pred": show_pred}


def build_views(sequence, predictions, track_rows, index, settings):
    frames = sequence["frames"]
    truth = sequence["masks"]
    prediction = None if predictions is None else predictions[index]
    base = viz.enhance(frames[index], settings["mode"], settings["cmap"], settings["denoise"])

    view1 = base.copy()
    if settings["show_truth"] and truth is not None:
        view1 = viz.overlay_mask(view1, truth[index], viz.TRUTH_COLOR)
    if settings["show_pred"] and prediction is not None:
        view1 = viz.overlay_mask(view1, prediction, viz.PREDICTION_COLOR)

    view2 = viz.temporal_difference(frames, index, settings["diff_ref"])

    view3 = base.copy()
    if prediction is not None:
        view3 = viz.overlay_mask(view3, prediction, viz.PREDICTION_COLOR)
    view3 = viz.draw_tracks(view3, track_rows, index)

    center = viz.centroid(prediction)
    if center is None and truth is not None:
        center = viz.centroid(truth[index])
    view4, _roi = viz.roi_zoom(view1, center, factor=4)
    return [view1, view2, view3, view4], center


def _run(commands, key):
    lines, box = [], st.empty()

    def emit(text):
        lines.append(str(text))
        box.code("\n".join(lines[-60:]))

    ok = cmd.run_commands(commands, scan.REPO_ROOT, emit)
    if ok:
        st.success("运行完成，结果已写入 runs/gui/ 下的新目录。")
    else:
        st.error("命令执行失败，请查看上方实时日志；不会覆盖任何已有实验存档。")
    return ok


def _pick_dir(candidates, label, key, preferred=None):
    options = list(dict.fromkeys([str(path) for path in ([preferred] if preferred else []) + list(candidates)]))
    labels = ["（未选择）"] + options
    index = 1 if options else 0
    chosen = st.selectbox(label, range(len(labels)), index=index,
                          format_func=lambda i: labels[i], key=f"{key}_select")
    manual = st.text_input(f"{label} · 或手动填写绝对路径", key=f"{key}_manual",
                           placeholder=r"例如 E:\workspace\weiboguangzi\runs\live-detect")
    if manual.strip():
        return Path(manual.strip())
    return Path(options[chosen - 1]) if chosen else None


def render_browser(sequence_dir, results, settings, key):
    """Four-panel frame browser plus dynamic KPI cards and track charts."""
    sequence = scan.load_sequence(sequence_dir)
    if sequence is None:
        st.error(f"无法加载序列：{sequence_dir}（需要 frames.npy 与 metadata.json）")
        return
    frames = sequence["frames"]
    count = len(frames)
    predictions = scan.load_predictions(results.get("predictions"), frames.shape)
    tracks = scan.load_tracks(results.get("tracks"))
    metrics = scan.load_metrics(results.get("metrics"))

    missing = [name for name, folder in (("predictions.npy", results.get("predictions")),
                                         ("tracks.csv", results.get("tracks")),
                                         ("metrics.json", results.get("metrics"))) if not folder]
    if missing or metrics is None:
        st.info("缺少 " + "、".join(missing or ["metrics.json"]) +
                "：相关指标显示为 —。可在模式 B 现场运行完整链路，或用上方输入框手动指定目录。")

    st.markdown("#### 关键指标（全部从 metrics.json / tracks.json 动态读取）")
    cards = scan.metric_cards(metrics, count)
    columns = st.columns(5)
    for position, (label, value) in enumerate(cards):
        columns[position % 5].metric(label, scan.format_metric(value))
    summary = scan.tracking_summary(metrics, tracks)
    columns = st.columns(3)
    for position, (label, value) in enumerate(summary.items()):
        columns[position % 3].metric(label, scan.format_metric(value))
    if metrics and metrics.get("match_iou") is not None:
        st.caption(f"目标级匹配阈值 IoU ≥ {scan.format_metric(metrics['match_iou'])}；"
                   f"指标来源：{metrics['path']}")

    st.markdown("#### 四宫格同步帧浏览器")
    st.caption("视图 1 增强 SAR 原图（绿色真值 / 红色预测轮廓） · 视图 2 时序差分（静止雪花与静态暗区被抵消） · "
               "视图 3 检测与轨迹（青色 measured / 橙色 imputed） · 视图 4 跟随阴影质心的 4 倍 ROI 放大")
    index = st.slider("统一帧号", 0, max(0, count - 1), 0, key=f"{key}_frame") if count > 1 else 0
    views, center = build_views(sequence, predictions, tracks["rows"], index, settings)
    st.image(viz.compose_quad(views),
             caption=f"帧 {index} / {count - 1}" + (f" · 质心 ({center[0]:.1f}, {center[1]:.1f})" if center else " · 本帧无阴影质心"),
             **_WIDTH_KWARG)

    control_a, control_b, control_c = st.columns([2, 2, 3])
    with control_a:
        play = st.button("播放四宫格动画", key=f"{key}_play")
    with control_b:
        export_gif = st.button("导出 GIF 动图", key=f"{key}_gif")
    with control_c:
        export_mp4 = st.button("导出 MP4（需 imageio-ffmpeg）", key=f"{key}_mp4")

    if play:
        placeholder = st.empty()
        for frame_index in range(count):
            panel_views, _ = build_views(sequence, predictions, tracks["rows"], frame_index, settings)
            placeholder.image(viz.compose_quad(panel_views), caption=f"播放中：帧 {frame_index} / {count - 1}",
                              **_WIDTH_KWARG)
            time.sleep(0.15)
        st.success("播放结束。")

    if export_gif or export_mp4:
        output = cmd.gui_run_dir("export")
        with st.spinner("正在生成四宫格动画 ..."):
            panels = [viz.compose_quad(build_views(sequence, predictions, tracks["rows"], i, settings)[0])
                      for i in range(count)]
            if export_gif:
                path = viz.save_gif(panels, output / "quad.gif", duration_ms=125)
                st.success(f"已保存：{path}")
                st.download_button("下载四宫格 GIF", path.read_bytes(),
                                   file_name="sar_quad.gif", mime="image/gif", key=f"{key}_dl_gif")
            else:
                try:
                    path = viz.save_mp4(panels, output / "quad.mp4", fps=8)
                    st.success(f"已保存：{path}")
                    st.download_button("下载四宫格 MP4", path.read_bytes(),
                                       file_name="sar_quad.mp4", mime="video/mp4", key=f"{key}_dl_mp4")
                except (ImportError, OSError, RuntimeError) as error:
                    st.error(f"MP4 导出失败：{error}。请安装 imageio-ffmpeg 或改用 GIF。")

    if tracks["rows"]:
        st.markdown("#### 逐帧轨迹坐标与表观速度")
        frame_indices, speeds = scan.per_frame_speed(tracks["rows"])
        coordinates = pd.DataFrame({
            "帧号": frame_indices,
            "x_px": [row["x_px"] for row in tracks["rows"]],
            "y_px": [row["y_px"] for row in tracks["rows"]],
        }).set_index("帧号")
        st.line_chart(coordinates, height=180)
        st.line_chart(pd.DataFrame({"帧号": frame_indices, "表观速度 (px/s)": speeds}).set_index("帧号"),
                      height=160)
        st.dataframe(pd.DataFrame(tracks["rows"]), **_WIDTH_KWARG)
        st.caption(f"来源：{tracks.get('path')}")


@st.cache_data(ttl=15, show_spinner=False)
def _sequences_cached(root_text):
    return [str(path) for path in scan.find_sequences(Path(root_text))]


@st.cache_data(ttl=15, show_spinner=False)
def _results_cached(root_text, sequence_text):
    raw = scan.find_results(Path(root_text), Path(sequence_text))
    return {kind: ([str(path) for path in value] if isinstance(value, list) else value)
            for kind, value in raw.items()}


def sequences_for(runs_path):
    """Cached scan, so a large runs/ tree is not walked on every interaction."""
    return [Path(text) for text in _sequences_cached(str(runs_path))]


def _sequence_label(path, runs_root):
    text = scan._relative_text(path, runs_root)
    if text.endswith(scan.PREFERRED_SEQUENCE):
        return f"[结题演示·优先] {text}"
    if "gui/" in text or text.startswith("gui"):
        return f"[GUI 现场结果] {text}"
    return text


def _results_for(runs_path, sequence_dir):
    with st.spinner("正在扫描该序列的结果目录 ..."):
        raw = _results_cached(str(runs_path), str(sequence_dir))
    results = {kind: ([Path(text) for text in value] if isinstance(value, list) else value)
               for kind, value in raw.items()}
    live = scan.live_dirs(runs_path)
    preferred = {"predictions": live.get("live-detect"),
                 "tracks": live.get("live-track"), "metrics": live.get("live-eval")}
    for kind, folder in preferred.items():
        if folder and folder.is_dir():
            results[kind] = [folder] + [path for path in results[kind] if path != folder]
    return results


def tab1_mode_a(runs_path, settings):
    st.markdown("##### 模式 A · 秒播已有结果（默认）")
    sequences = sequences_for(runs_path)
    manual = st.text_input("手动指定序列目录（绝对路径，留空则用下方扫描结果）", key="a_manual_seq")
    if manual.strip():
        sequence_dir = Path(manual.strip())
    elif sequences:
        choice = st.selectbox("扫描到的 frames.npy 数据目录", range(len(sequences)),
                              format_func=lambda i: _sequence_label(sequences[i], runs_path), key="a_seq")
        sequence_dir = sequences[choice]
    else:
        st.warning(f"未在 {runs_path} 找到包含 frames.npy 的目录（该目录可能被 .gitignore 排除或尚未生成）。")
        st.markdown("处理方式：① 在上方手动填入已有结果路径；② 一键生成快速演示；③ 切到模式 B 现场运行。")
        if st.button("一键生成快速图像域演示", key="a_bootstrap"):
            output = cmd.gui_run_dir("bootstrap")
            demo_dir = output / "demo"
            if _run([cmd.demo_command(cmd.DEMO_CONFIG, 16, demo_dir)], "a_boot"):
                render_browser(demo_dir / "sequence",
                               {"predictions": demo_dir / "detect", "tracks": demo_dir / "track",
                                "metrics": demo_dir / "evaluate"}, settings, "a_boot_browser")
        return

    if not sequence_dir.is_dir() or not (sequence_dir / "frames.npy").is_file():
        st.error(f"目录无效或缺少 frames.npy：{sequence_dir}")
        return
    results = _results_for(runs_path, sequence_dir)
    if not any(results["_linked"].values()):
        st.caption("未找到与所选序列通过 source.json 关联的结果目录，以下为全部候选，请手动核对。"
                   "如已按 README 生成 runs/live-detect / live-track / live-eval，它们已优先置顶。")
    with st.expander("结果目录（检测 / 跟踪 / 评估）", expanded=True):
        left, middle, right = st.columns(3)
        with left:
            detect_dir = _pick_dir(results["predictions"], "检测结果目录", "a_detect")
        with middle:
            track_dir = _pick_dir(results["tracks"], "跟踪结果目录", "a_track")
        with right:
            evaluate_dir = _pick_dir(results["metrics"], "评估结果目录", "a_eval")
    render_browser(sequence_dir, {"predictions": detect_dir, "tracks": track_dir,
                                  "metrics": evaluate_dir}, settings, "tab1_a")


def tab1_mode_b(settings):
    st.markdown("##### 模式 B · 现场实时运行")
    st.caption("所有现场产物写入 runs/gui/<时间戳_任务名>/，绝不覆盖已有 runs/ 存档。")
    choice = st.radio("运行方式", ("快速图像域演示（推荐现场用）", "单通道 SAR 回波成像与检测"),
                      key="b_choice")
    if choice.startswith("快速"):
        config = st.text_input("配置 JSON", value=str(cmd.DEMO_CONFIG), key="b_config")
        frames = st.slider("帧数", 8, 24, 16, step=1, key="b_frames")
        st.code(f"{Path(sys.executable).name} -m videosar demo --config {config} --frames {frames}")
        if st.button("运行快速演示", key="b_run_demo"):
            output = cmd.gui_run_dir("demo")
            demo_dir = output / "demo"
            if _run([cmd.demo_command(config, frames, demo_dir)], "b_demo_log"):
                st.markdown("##### 运行结果 · 四宫格浏览器")
                render_browser(demo_dir / "sequence",
                               {"predictions": demo_dir / "detect", "tracks": demo_dir / "track",
                                "metrics": demo_dir / "evaluate"}, settings, "b_demo_browser")
    else:
        bandwidth = st.selectbox("带宽 (bandwidth-hz)", (300000000, 450000000, 600000000),
                                 format_func=lambda value: f"{value / 1e6:.0f} MHz", key="b_bw")
        method = st.selectbox("检测方法", ("threshold", "difference", "learned"),
                              format_func=lambda value: {"threshold": "传统：局部暗度阈值",
                                                         "difference": "传统：时序背景差分",
                                                         "learned": "学习模型：冻结 U-Net（需 torch）"}[value],
                              key="b_method")
        model = cmd.UNET_MODEL
        if method == "learned":
            model = Path(st.text_input("模型部署 JSON", value=str(cmd.UNET_MODEL), key="b_model"))
        st.code("image -> detect -> track -> evaluate（--match-iou 0.50）")
        if st.button("运行 SAR 回波成像与检测全链路", key="b_run_sar"):
            output = cmd.gui_run_dir("sar")
            plan = cmd.sar_chain_commands(bandwidth, method, model, output)
            if _run(plan["commands"], "b_sar_log"):
                st.markdown("##### 运行结果 · 四宫格浏览器")
                render_browser(plan["sequence"], plan, settings, "b_sar_browser")

def _numeric_frame(rows):
    frame = pd.DataFrame(rows)
    for column in frame.columns:
        converted = pd.to_numeric(frame[column], errors="coerce")
        if converted.notna().sum() >= max(1, int(0.8 * len(frame))):
            frame[column] = converted
    return frame


def render_tab2(runs_path, settings):
    st.subheader("Tab 2 · 算法与物理条件同屏对照")
    sequences = sequences_for(runs_path)
    if not sequences:
        st.warning(f"未在 {runs_path} 找到 frames.npy 序列，请先在 Tab 1 生成结果或手动填写路径。")
        manual = st.text_input("序列目录绝对路径", key="t2_manual_seq")
        if not manual.strip():
            return
        sequences = [Path(manual.strip())]
    choice = st.selectbox("同源 frames.npy 序列", range(len(sequences)),
                          format_func=lambda i: _sequence_label(sequences[i], runs_path), key="t2_seq")
    sequence_dir = sequences[choice]
    sequence = scan.load_sequence(sequence_dir)
    if sequence is None:
        st.error(f"无法加载序列：{sequence_dir}")
        return
    frames = sequence["frames"]

    results = _results_for(runs_path, sequence_dir)
    candidates = [str(path) for path in results["predictions"]]
    candidates += [path for path in st.session_state.get("t2_extra", []) if path not in candidates]
    if st.button("对当前序列一键跑：传统基线 vs 学习模型", key="t2_compare_run"):
        output = cmd.gui_run_dir("compare")
        plan = cmd.compare_commands(sequence_dir, output)
        if _run(plan["commands"], "t2_compare_log"):
            st.session_state["t2_extra"] = [str(path) for path in plan["predictions"]]
            candidates = [str(path) for path in plan["predictions"]] + candidates
    if not candidates:
        st.info("未找到任何 predictions.npy 目录。可点击上方按钮现跑两组检测，或在 Tab 1 模式 B 生成。")
        return

    left_col, right_col = st.columns(2)
    with left_col:
        left = st.selectbox("左侧检测目录", candidates, key="t2_left")
    with right_col:
        right = st.selectbox("右侧检测目录", candidates, index=min(1, len(candidates) - 1), key="t2_right")
    index = st.slider("同屏帧号", 0, max(0, len(frames) - 1), 0, key="t2_frame") if len(frames) > 1 else 0
    truth = sequence["masks"]
    base = viz.enhance(frames[index], settings["mode"], settings["cmap"], settings["denoise"])

    for column, path, title in zip((left_col, right_col), (left, right), ("左侧", "右侧")):
        prediction = scan.load_predictions(Path(path), frames.shape)
        view = base.copy()
        if prediction is not None:
            view = viz.overlay_mask(view, prediction[index], viz.PREDICTION_COLOR)
        if truth is not None:
            view = viz.overlay_mask(view, truth[index], viz.TRUTH_COLOR)
        column.image(view, caption=f"{title}：{path}", **_WIDTH_KWARG)
        components = viz.count_components(prediction[index]) if prediction is not None else None
        overlap = (int(np.count_nonzero(prediction[index] & truth[index]))
                   if prediction is not None and truth is not None else None)
        metric_a, metric_b = column.columns(2)
        metric_a.metric("预测连通域数（预览计数）", scan.format_metric(components))
        metric_b.metric("与真值重叠像素", scan.format_metric(overlap))
        column.caption("虚警数以对应 evaluate/metrics.json 为准；红色=预测，绿色=真值轮廓。")
    if left == right:
        st.caption("提示：左右选择了同一目录，可在上方改成两个不同的检测结果。")

    st.markdown("#### 归档指标表（自动从 deliverables/ 读取）")
    tables = scan.archive_tables()
    if not tables:
        st.info("未找到归档 CSV。")
        return
    labels = [label for label, _ in tables]
    selected = st.selectbox("选择归档指标表", range(len(tables)),
                            format_func=lambda i: labels[i], key="t2_table")
    label, path = tables[selected]
    fields, rows = scan.read_csv_rows(path)
    if not rows:
        st.warning(f"{path} 无可读取的数据行。")
        return
    frame = _numeric_frame(rows)
    st.caption(f"{label} · 来源：{path}")
    st.dataframe(frame, **_WIDTH_KWARG)
    try:
        if "condition" in frame.columns and "recall" in frame.columns and "method" in frame.columns:
            pivot = frame.pivot_table(index="condition", columns="method", values="recall", aggfunc="mean")
            st.markdown("分条件召回率对比")
            st.bar_chart(pivot)
        elif "method" in frame.columns:
            metrics = [name for name in ("precision", "recall", "target_f1", "region_iou",
                                         "false_positive_per_frame") if name in frame.columns]
            if metrics:
                st.bar_chart(frame.set_index("method")[metrics].apply(pd.to_numeric, errors="coerce"))
    except (KeyError, ValueError, TypeError) as error:
        st.caption(f"柱状图跳过：{error}")

IMPUTED_COLOR = (255, 202, 118)
TRUTH_COLOR = (64, 235, 120)
PREDICTION_COLOR = (255, 92, 92)
CROP_COLOR = (255, 64, 64)

MODES = ("percentile", "log_db", "linear")
MODE_LABELS = {

# Approximate matplotlib magma / viridis anchors, so no matplotlib dependency
# is needed. Interpolated to a 256-entry LUT below.
_ANCHORS = {
    "gray": [(0, 0, 0), (255, 255, 255)],
    "magma": [(0, 0, 4), (40, 17, 76), (115, 25, 105), (188, 55, 105),
              (245, 108, 81), (252, 190, 77), (252, 253, 191)],
    "viridis": [(68, 1, 84), (72, 40, 120), (62, 74, 137), (49, 104, 142),
                (38, 130, 142), (53, 183, 121), (181, 222, 43)],
}
_DIVERGING_ANCHORS = np.array([[40, 90, 170], [14, 18, 24], [255, 176, 64]], dtype=np.float64)


def _lut(anchors, size=256):
    points = np.asarray(anchors, dtype=np.float64)
    positions = np.linspace(0, size - 1, len(points))
    grid = np.arange(size)
    channels = [np.interp(grid, positions, points[:, c]) for c in range(3)]
    return np.rint(np.stack(channels, axis=1)).astype(np.uint8)


LUTS = {name: _lut(points) for name, points in _ANCHORS.items()}
DIVERGING = _lut(_DIVERGING_ANCHORS)


def smooth(frame):
    """Mild 3x3 box despeckle with edge padding (preview only)."""
    image = np.asarray(frame, dtype=np.float32)
    padded = np.pad(image, 1, mode="edge")
    total = np.zeros_like(image)
    for dy in range(3):
        for dx in range(3):
            total += padded[dy:dy + image.shape[0], dx:dx + image.shape[1]]
    return total / 9.0


def normalize(frame, mode="percentile"):
    """Map a nonnegative intensity frame to [0, 1] float32."""
    image = np.asarray(frame, dtype=np.float64)
    if mode == "log_db":
        peak = float(image.max()) if image.size else 0.0
        reference = peak if peak > 0 else 1.0
        image = 20.0 * np.log10(np.maximum(image / reference, 1e-4))
        low, high = -40.0, 0.0
    elif mode == "linear":
        low, high = 0.0, float(image.max()) if image.size else 1.0
    else:
        if image.size:
            low, high = (float(value) for value in np.percentile(image, [2, 98]))
        else:
            low, high = 0.0, 1.0
    if high <= low:
        low, high = float(low) - 1.0, float(low) + 1.0
    return np.clip((image - low) / (high - low), 0.0, 1.0).astype(np.float32)


def colorize(normalized, cmap="gray"):
    """Apply a 256-colour LUT to a [0, 1] frame, returning RGB uint8."""
    index = np.clip(np.rint(np.asarray(normalized, dtype=np.float32) * 255), 0, 255).astype(np.uint8)
    return LUTS.get(cmap, LUTS["gray"])[index].copy()


def enhance(frame, mode="percentile", cmap="gray", denoise=False):
    """One-call display enhancement; never used as detector input."""
    image = np.asarray(frame, dtype=np.float32)
    if denoise:
        image = smooth(image)
    return colorize(normalize(image, mode), cmap)


def temporal_difference(frames, index, reference="median", window=5):
    """Centered diverging RGB view of the change at one frame.

    Static speckle and static dark regions cancel between current and history,
    while a moving shadow produces a strong positive (warm) response.
    """
    current = np.asarray(frames[index], dtype=np.float32)
    if reference == "previous":
        baseline = np.asarray(frames[max(0, index - 1)], dtype=np.float32)
    else:
        start = max(0, index - max(1, int(window)))
        history = np.asarray(frames[start:index], dtype=np.float32)
        baseline = np.median(history, axis=0) if len(history) else current
    delta = baseline - current
    scale = float(np.percentile(np.abs(delta), 98)) if delta.size else 0.0
    scale = max(scale, 1e-6)
    centered = np.clip(delta / (2.0 * scale) + 0.5, 0.0, 1.0)
    lookup = np.clip(np.rint(centered * 255), 0, 255).astype(np.uint8)
    return DIVERGING[lookup].copy()


def boundary(mask):
    """One-pixel outline of a binary mask (four-connected erosion)."""
    binary = np.asarray(mask).astype(bool)
    if not binary.any():
        return np.zeros_like(binary)
    padded = np.pad(binary, 1, mode="constant", constant_values=False)
    inner = (binary & padded[:-2, 1:-1] & padded[2:, 1:-1]
             & padded[1:-1, :-2] & padded[1:-1, 2:])
    return binary & ~inner


def overlay_mask(rgb, mask, color, alpha=0.9):
    """Draw a thin coloured contour over an RGB image."""
    if mask is None:
        return rgb
    output = np.asarray(rgb).copy()
    edge = boundary(mask)
    if not edge.any():
        return output
    tint = np.asarray(color, dtype=np.float32)
    output[edge] = np.rint(output[edge] * (1.0 - alpha) + tint * alpha).astype(np.uint8)
    return output


def _point(row):
    x, y = row.get("x_px"), row.get("y_px")
    if x is None or y is None or x == "" or y == "":
        return None
    try:
        return float(x), float(y)
    except (TypeError, ValueError):
        return None


def draw_tracks(rgb, rows, frame_index, radius=2):
    """Draw the track history up to ``frame_index`` using repository colours."""
    image = Image.fromarray(np.asarray(rgb, dtype=np.uint8))
    painter = ImageDraw.Draw(image)
    last = None
    for row in rows:
        try:
            index = int(row["frame"])
        except (KeyError, TypeError, ValueError):
            continue
        if index > frame_index:
            break
        point = _point(row)
        if point is None:
            last = None
            continue
        color = IMPUTED_COLOR if row.get("source") == "imputed" else MEASURED_COLOR
        segment = row.get("segment")
        if last is not None and last["segment"] == segment:
            painter.line([last["point"], point], fill=color, width=1)
        if index == frame_index:
            x, y = point
            painter.ellipse((x - radius, y - radius, x + radius, y + radius), outline=color)
        last = {"point": point, "segment": segment}
    return np.asarray(image)


def centroid(mask):
    """Return (x, y) centre of a binary mask, or None when empty."""
    if mask is None:
        return None
    binary = np.asarray(mask).astype(bool)
    if not binary.any():
        return None
    y, x = np.nonzero(binary)
    return float(x.mean()), float(y.mean())


def roi_zoom(rgb, center, factor=4, out_size=(256, 192)):
    """Crop around ``center`` and upscale with nearest-neighbour (4x by default)."""
    image = np.asarray(rgb, dtype=np.uint8)
    height, width = image.shape[:2]
    if center is None:
        center = (width / 2.0, height / 2.0)
    crop_width = max(8, int(round(out_size[0] / max(1, factor))))
    crop_height = max(8, int(round(out_size[1] / max(1, factor))))
    x0 = int(np.clip(round(center[0] - crop_width / 2.0), 0, max(0, width - crop_width)))
    y0 = int(np.clip(round(center[1] - crop_height / 2.0), 0, max(0, height - crop_height)))
    patch = image[y0:y0 + crop_height, x0:x0 + crop_width]
    zoomed = Image.fromarray(patch).resize(out_size, Image.Resampling.NEAREST)
    # Mark the tracked centre inside the magnified view.
    painter = ImageDraw.Draw(zoomed)
    marker_x = (center[0] - x0) * factor
    marker_y = (center[1] - y0) * factor
    if 0 <= marker_x < out_size[0] and 0 <= marker_y < out_size[1]:
        painter.line((marker_x - 5, marker_y, marker_x + 5, marker_y), fill=MEASURED_COLOR)
        painter.line((marker_x, marker_y - 5, marker_x, marker_y + 5), fill=MEASURED_COLOR)
    return np.asarray(zoomed), (x0, y0, crop_width, crop_height)


def count_components(mask):
    """Four-connected component count, mirroring videosar.analysis.components."""
    binary = np.asarray(mask).astype(bool)
    height, width = binary.shape
    seen = np.zeros_like(binary)
    count = 0
    for y in range(height):
        for x in range(width):
            if not binary[y, x] or seen[y, x]:
                continue
            count += 1
            stack = [(y, x)]
            seen[y, x] = True
            while stack:
                cy, cx = stack.pop()
                for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                    if 0 <= ny < height and 0 <= nx < width and binary[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        stack.append((ny, nx))
    return count


PANEL_TITLES = ("1  Enhanced SAR", "2  Temporal difference", "3  Detections + track", "4  ROI zoom x4")


def compose_quad(views, titles=PANEL_TITLES, gap=8, bar=22):
    """Compose four RGB views into one labelled 2x2 panel."""
    panels = [np.asarray(view, dtype=np.uint8) for view in views]
    size = (max(view.shape[1] for view in panels), max(view.shape[0] for view in panels))
    canvas = Image.new("RGB", (2 * size[0] + 3 * gap, 2 * (size[1] + bar) + 3 * gap), (18, 27, 34))
    painter = ImageDraw.Draw(canvas)
    for index, (view, title) in enumerate(zip(panels, titles, strict=True)):
        column, row = index % 2, index // 2
        x = gap + column * (size[0] + gap)
        y = gap + row * (size[1] + bar + gap)
        painter.text((x + 2, y + 4), title, fill=(230, 238, 244))
        panel = Image.fromarray(view).resize(size, Image.Resampling.NEAREST)
        canvas.paste(panel, (x, y + bar))
    return np.asarray(canvas)


def save_gif(frames, path, duration_ms=120):
    """Write an animated GIF using Pillow only."""
    from pathlib import Path

    images = [Image.fromarray(np.asarray(frame, dtype=np.uint8)) for frame in frames]
    if not images:
        return None
    destination = Path(path)
    destination.parent.mkdir(parents=True, exist_ok=True)
    images[0].save(destination, save_all=True, append_images=images[1:],
                   duration=duration_ms, loop=0)
    return destination


def save_mp4(frames, path, fps=8):
    """Optional MP4 export; requires imageio-ffmpeg (already in requirements-gui.txt)."""
    from pathlib import Path

    import imageio_ffmpeg

    panels = [np.asarray(frame, dtype=np.uint8) for frame in frames]
    if not panels:
        return None
    height, width = panels[0].shape[:2]
    even = (width // 2 * 2, height // 2 * 2)
    destination = Path(path)
    destination.parent.mkdir(parents=True, exist_ok=True)
    writer = imageio_ffmpeg.write_frames(str(destination), even, fps=fps, codec="libx264",
                                         pix_fmt_in="rgb24", pix_fmt_out="yuv420p",
                                         output_params=["-movflags", "+faststart"])
    writer.send(None)
    try:
        for panel in panels:
            writer.send(np.ascontiguousarray(panel[:even[1], :even[0]]).tobytes())
    finally:
        writer.close()
    return destination

def read_json_safe(path):
    """Return a JSON object, or None when missing / malformed."""
    try:
        with Path(path).open(encoding="utf-8-sig") as stream:
            return json.load(stream)
    except (OSError, ValueError, TypeError):
        return None


def read_csv_rows(path):
    """Return (fieldnames, rows). Absent or broken files return (None, [])."""
    source = Path(path)
    if not source.is_file():
        return None, []
    try:
        with source.open(encoding="utf-8-sig", newline="") as stream:
            reader = csv.DictReader(stream)
            return reader.fieldnames, list(reader)
    except (OSError, UnicodeError, csv.Error):
        return None, []


def numeric(value):
    """Convert a CSV cell to float, tolerating blanks and text."""
    if value is None or value == "":
        return None
    try:
        return float(value)
    except (TypeError, ValueError):
        return None


def format_metric(value):
    """Format a metric for a card; '—' when unavailable."""
    if value is None:
        return "—"
    if isinstance(value, bool):
        return str(value)
    if isinstance(value, int):
        return str(value)
    try:
        return f"{float(value):.3f}"
    except (TypeError, ValueError):
        return str(value)


def _relative_text(path, root):
    try:
        return str(path.relative_to(root)).replace("\\", "/")
    except ValueError:
        return str(path).replace("\\", "/")


def _sequence_rank(path, root):
    text = _relative_text(path, root).lower()
    if text.endswith(PREFERRED_SEQUENCE):
        return 0
    if text.startswith("gui/") or "/gui/" in text:
        return 1
    if any(name in text for name in PREFERRED_LIVE):
        return 2
    return 3


def find_sequences(runs_root, max_results=250):
    """Find directories containing frames.npy, preferred demo data first."""
    root = Path(runs_root)
    found, seen = [], set()
    if not root.is_dir():
        return found
    for dirpath, _dirnames, filenames in os.walk(root):
        if "frames.npy" not in filenames:
            continue
        candidate = Path(dirpath)
        key = str(candidate).lower()
        if key in seen:
            continue
        seen.add(key)
        found.append(candidate)
    # Sort the complete set before truncating: an early break would drop the
    # preferred closeout demo whenever runs/ contains many newer sequences.
    found.sort(key=lambda path: (_sequence_rank(path, root), -path.stat().st_mtime, str(path)))
    return found[:max_results]


def load_sequence(directory):
    """Load frames, optional truth masks and metadata from a sequence folder."""
    root = Path(directory)
    frames_path = root / "frames.npy"
    if not frames_path.is_file():
        return None
    try:
        frames = np.asarray(np.load(frames_path, allow_pickle=False), dtype=np.float32)
    except (OSError, ValueError):
        return None
    if frames.ndim != 3 or frames.size == 0:
        return None
    masks = None
    masks_path = root / "masks.npy"
    if masks_path.is_file():
        try:
            candidate = np.load(masks_path, allow_pickle=False)
            if candidate.shape == frames.shape:
                masks = candidate.astype(bool)
        except (OSError, ValueError):
            masks = None
    return {"root": root, "frames": frames, "masks": masks,
            "metadata": read_json_safe(root / "metadata.json") or {}}


def _linked_to(directory, target):
    receipt = read_json_safe(Path(directory) / "source.json")
    if not isinstance(receipt, dict):
        return False
    sequence = receipt.get("sequence")
    if not sequence:
        return False
    try:
        return Path(sequence).resolve() == Path(target).resolve()
    except (OSError, ValueError):
        return False


def find_results(runs_root, sequence_root, max_each=60):
    """Locate detect / track / evaluate folders for one sequence.

    A source.json receipt links a result folder to its input sequence. When no
    receipt matches, all candidates are still returned so the user can pick
    manually instead of hitting an error.
    """
    root = Path(runs_root)
    buckets = {"predictions": [], "tracks": [], "metrics": [], "_linked": {}}
    if not root.is_dir():
        return buckets
    for dirpath, _dirnames, filenames in os.walk(root):
        folder = Path(dirpath)
        if "predictions.npy" in filenames:
            buckets["predictions"].append(folder)
        if "tracks.csv" in filenames:
            buckets["tracks"].append(folder)
        if "metrics.json" in filenames:
            buckets["metrics"].append(folder)
    target = Path(sequence_root)
    for kind in ("predictions", "tracks", "metrics"):
        candidates = buckets[kind]
        linked = [path for path in candidates if _linked_to(path, target)]
        buckets["_linked"][kind] = bool(linked)
        chosen = linked if linked else candidates
        buckets[kind] = sorted(chosen, key=lambda path: -path.stat().st_mtime)[:max_each]
    return buckets


def load_predictions(detect_dir, expected_shape):

    return masks.astype(bool) if masks.shape == expected_shape else None

def per_frame_speed(rows):
    """Apparent per-frame speed (px/s) computed from consecutive track points."""
    speeds, indices = [], []
    previous = None
    for row in rows:
        x, y, time = row.get("x_px"), row.get("y_px"), row.get("time_s")
        if x is None or y is None or time is None:
            previous = None
            speeds.append(None)
            indices.append(row["frame"])
            continue
        if previous is not None and time > previous[2]:
            speeds.append(float(np.hypot(x - previous[0], y - previous[1]) / (time - previous[2])))
        else:
            speeds.append(None)
        indices.append(row["frame"])
        previous = (x, y, time)
    return indices, speeds

def run_commands(commands, cwd, emit):
    """Run commands in order, streaming combined stdout/stderr to ``emit``."""
    for command in commands:
        emit("> " + " ".join(command[2:]))
        try:
            process = subprocess.Popen(
                command, cwd=str(cwd), stdout=subprocess.PIPE, stderr=subprocess.STDOUT,
                text=True, encoding="utf-8", errors="replace", bufsize=1)
        except OSError as error:
            emit(f"[启动失败] {error}")
            return False
        with process:  # Popen.__exit__ closes the pipe and waits for the child
            assert process.stdout is not None
            for line in process.stdout:
                emit(line.rstrip())
            code = process.wait()
        emit(f"[退出码 {code}]")
        if code != 0:
            return False
    return True

def extract_first_frame(video, ffmpeg, destination):
    """Decode the first video frame to a PNG for the --crop preview."""
    target = Path(destination)
    target.parent.mkdir(parents=True, exist_ok=True)
    command = [str(ffmpeg), "-nostdin", "-v", "error", "-i", str(video),
               "-frames:v", "1", "-y", str(target)]
    try:
        result = subprocess.run(command, capture_output=True, text=True, timeout=90)
    except (OSError, subprocess.SubprocessError):
        return None
    if result.returncode != 0 or not target.is_file():
        return None
    return target

class CommandsTest(TestCase):
    def test_demo_and_sar_commands_use_real_flag_names(self):
        demo = cmd.demo_command(cmd.DEMO_CONFIG, 12, Path("runs/gui/demo"))
        self.assertEqual(demo[2:4], ["videosar", "demo"])
        self.assertIn("--config", demo)
        self.assertIn("--frames", demo)
        self.assertIn("--output-dir", demo)
        plan = cmd.sar_chain_commands(450000000, "learned", cmd.UNET_MODEL, Path("runs/gui/sar"))
        self.assertEqual(len(plan["commands"]), 4)
        joined = [part for command in plan["commands"] for part in command]
        for flag in ("--bandwidth-hz", "--input", "--method", "--model", "--predictions",
                     "--tracks", "--match-iou"):
            self.assertIn(flag, joined)
        self.assertIn("0.50", joined)
        traditional = cmd.sar_chain_commands(300000000, "threshold", None, Path("runs/gui/sar2"))
        self.assertNotIn("--model", [part for command in traditional["commands"] for part in command])

    def test_detect_video_and_compare_commands(self):
        command = cmd.detect_video_command(Path("in.mp4"), Path("runs/gui/v"), crop=(10, 20, 300, 400),
                                           start_frame=150, max_frames=24)
        self.assertEqual(command[2:4], ["videosar", "detect-video"])
        self.assertEqual(command[command.index("--crop") + 1:command.index("--crop") + 5],
                         ["10", "20", "300", "400"])
        self.assertIn("--start-frame", command)
        self.assertIn("--max-frames", command)
        self.assertIn("--output-dir", command)
        self.assertNotIn("--model", command)
        comparison = cmd.compare_commands(Path("dataset/s1"), Path("runs/gui/cmp"))
        self.assertEqual(len(comparison["commands"]), 2)
        self.assertIn("--method", comparison["commands"][0])
        self.assertIn("learned", comparison["commands"][1])

    def test_gui_run_dir_is_under_runs_gui(self):
        with TemporaryDirectory() as folder:
            path = cmd.gui_run_dir("demo", Path(folder))
            self.assertEqual(path.parent.name, "gui")
            self.assertEqual(path.parent.parent.name, "runs")
            self.assertTrue(path.name.endswith("_demo"))

    def test_run_commands_streams_output_and_reports_failure(self):
        with TemporaryDirectory() as folder:
            lines = []
            ok = cmd.run_commands([[sys.executable, "-c", "print('hello-gui')"]], folder, lines.append)
            self.assertTrue(ok)
            self.assertIn("hello-gui", "\n".join(lines))
            self.assertFalse(cmd.run_commands([[sys.executable, "-c", "raise SystemExit(3)"]],
                                              folder, lines.append))
            self.assertTrue(any("退出码 3" in line for line in lines))

    def test_environment_report_keys(self):
        report = cmd.environment_report()
        for key in ("ffmpeg", "torch", "torchvision", "imageio", "imageio_ffmpeg", "python"):
            self.assertIn(key, report)
        self.assertTrue(report["python"])
